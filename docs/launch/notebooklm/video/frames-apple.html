SELECT p.sku, p.name, SUM(oi.qty) AS u, SUM(oi.qty*oi.price) AS rev
FROM orders o
JOIN order_items oi ON oi.order_id=o.id
JOIN products p ON p.sku=oi.sku
WHERE o.paid_at >= NOW()-'7d'::interval
GROUP BY p.sku, p.name ORDER BY rev DESC LIMIT 10;
